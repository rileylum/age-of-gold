cmake_minimum_required(VERSION 3.20)
project(AgeOfGoldShaderExecution LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  message(FATAL_ERROR "Use the pinned LLVM-MinGW Windows compiler")
endif()
add_executable(aog-shader-execution main.cpp)
target_compile_features(aog-shader-execution PRIVATE cxx_std_17)
target_link_libraries(aog-shader-execution PRIVATE d3d12 d3dcompiler dxgi ole32)
if(MINGW)
  target_link_options(aog-shader-execution PRIVATE -static -Wl,--no-insert-timestamp)
endif()
