cmake_minimum_required(VERSION 3.20)
project(AgeOfGoldNativeChecks LANGUAGES C CXX)
if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  message(FATAL_ERROR "Use a generator that honours the pinned LLVM-MinGW compiler")
endif()

if(NOT WIN32)
  message(FATAL_ERROR "The lighting runtime targets Windows; configure a Windows compiler")
endif()
set(AOG_DEPENDENCIES "$ENV{AOG_DEPENDENCIES}" CACHE PATH "External pinned dependency checkouts")
if(NOT EXISTS "${AOG_DEPENDENCIES}/MinHook/CMakeLists.txt")
  message(FATAL_ERROR "Run eng/bootstrap.py outside the repository and set AOG_DEPENDENCIES")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build the hook dependency into the runtime" FORCE)
add_subdirectory("${AOG_DEPENDENCIES}/MinHook" minhook)
add_executable(aog-native-hooks check.cpp)
target_compile_features(aog-native-hooks PRIVATE cxx_std_17)
target_link_libraries(aog-native-hooks PRIVATE minhook)
if(MINGW)
  target_link_options(aog-native-hooks PRIVATE -static -Wl,--no-insert-timestamp)
endif()
configure_file("${AOG_DEPENDENCIES}/MinHook/LICENSE.txt" "LICENSES/MinHook.txt" COPYONLY)
configure_file("${AOG_LLVM_MINGW}/LICENSE.TXT" "LICENSES/LLVM-MinGW.txt" COPYONLY)
file(GLOB _aog_runtime_notices "${AOG_LLVM_MINGW}/x86_64-w64-mingw32/share/mingw32/COPYING*")
if(NOT _aog_runtime_notices)
  message(FATAL_ERROR "The external compiler is missing its runtime licence notices")
endif()
foreach(_aog_notice IN LISTS _aog_runtime_notices)
  get_filename_component(_aog_notice_name "${_aog_notice}" NAME)
  configure_file("${_aog_notice}" "LICENSES/mingw-w64/${_aog_notice_name}" COPYONLY)
endforeach()
enable_testing()
add_test(NAME hooks COMMAND aog-native-hooks)
add_executable(aog-host-state host_state.cpp ../../src/RuntimeNative/HostState.cpp)
target_include_directories(aog-host-state PRIVATE ../../src/RuntimeNative)
target_compile_features(aog-host-state PRIVATE cxx_std_17)
target_link_libraries(aog-host-state PRIVATE bcrypt)
if(MINGW)
  target_link_options(aog-host-state PRIVATE -static -Wl,--no-insert-timestamp)
endif()
